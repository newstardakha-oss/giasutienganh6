        function speakText(text) {
            if (!synth) return;
            synth.cancel();

            if (gameState.phrasePause && text.includes(",")) {
                const parts = text.split(/([,;.])/);
                let delay = 0;
                for (let i = 0; i < parts.length; i += 2) {
                    const chunk = parts[i];
                    if (chunk && chunk.trim().length > 0) {
                        setTimeout(() => {
                            let utt = new SpeechSynthesisUtterance(chunk);
                            applyVoiceSettings(utt);
                            synth.speak(utt);
                        }, delay);
                        delay += (chunk.length * 60) + 300;
                    }
                }
            } else {
                let utterance = new SpeechSynthesisUtterance(text);
                applyVoiceSettings(utterance);
                synth.speak(utterance);
            }
        }

            if (gameState.selectedVoiceURI) {
                const v = voices.find(item => item.voiceURI === gameState.selectedVoiceURI);
                if (v) utterance.voice = v;
            }
        }

        function shuffleArray(array) {
            let arr = [...array];
            for (let i = arr.length - 1; i > 0; i--) {
                const j = Math.floor(Math.random() * (i + 1));
                [arr[i], arr[j]] = [arr[j], arr[i]];
            }
            return arr;
        }

                            ${shuffledRights.map(r => `<div class="match-item" data-val="${r}" onclick="selectMatchItem(this, 'right')">${r}</div>`).join('')}
                        </div>
                    </div>
                `;
                area.innerHTML = html;
            } else if (type === 3) {
                let html = matchingData.type3.map(m => `
                    <div style="background:#fff; padding:15px; border-radius:8px; margin-bottom:10px;">
                        <p><strong>Cụm từ:</strong> "${m.phrase}"</p>
                        <p style="color:var(--secondary-dark);">➔ <strong>Đồng nghĩa:</strong> "${m.synonym}"</p>
                        <button class="btn btn-secondary btn-sm" onclick="speakText('${m.phrase}')">🔊 Nghe cụm</button>
                    </div>
                `).join('');
                area.innerHTML = html;
                gameState.completedTasks.matching = true;
                saveGameState();
            }
        }

        function checkMatchingType1(btn, correct, chosen) {
            if (chosen === correct) {
                btn.classList.add("correct");
                addScore(4);
                speakText(correct);
            } else {
                btn.classList.add("incorrect");
            }
        }

        function selectMatchItem(el, side) {
            if (el.classList.contains("matched")) return;

            if (side === 'left') {
                document.querySelectorAll("#match-col-left .match-item").forEach(i => i.classList.remove("selected"));
                el.classList.add("selected");
                selectedMatchLeft = el;
            } else {
                document.querySelectorAll("#match-col-right .match-item").forEach(i => i.classList.remove("selected"));
                el.classList.add("selected");
                selectedMatchRight = el;
            }

            if (selectedMatchLeft && selectedMatchRight) {
                const lVal = selectedMatchLeft.getAttribute("data-val");
                const rVal = selectedMatchRight.getAttribute("data-val");

                const pair = matchingData.type2.find(p => p.left === lVal);
                if (pair && pair.right === rVal) {
                    selectedMatchLeft.classList.remove("selected");
                    selectedMatchRight.classList.remove("selected");
                    selectedMatchLeft.classList.add("matched");
                    selectedMatchRight.classList.add("matched");
                    addScore(4);
                    speakText(lVal + " " + rVal);
                } else {
                    selectedMatchLeft.classList.remove("selected");
                    selectedMatchRight.classList.remove("selected");
                }
                selectedMatchLeft = null;
                selectedMatchRight = null;
            }
        }

        /* WRITING TASK */
        let currentScrambleSource = [];
        let currentScrambleTarget = [];

        function switchWritingMode(mode) {
            document.getElementById("writing-scramble-area").style.display = mode === 'scramble' ? 'block' : 'none';
            document.getElementById("writing-guided-area").style.display = mode === 'guided' ? 'block' : 'none';

            if (mode === 'scramble') initScrambleSentence();
            if (mode === 'guided') initGuidedWriting();
        }

        function initScrambleSentence() {
            const data = scrambleData[gameState.scrambleIndex];
            currentScrambleSource = shuffleArray(data.chunks);
            currentScrambleTarget = [];
            renderScrambleUI();
        }

        function renderScrambleUI() {
            const sourceBox = document.getElementById("scramble-source-box");
            const targetBox = document.getElementById("scramble-target-box");

            sourceBox.innerHTML = currentScrambleSource.map((chunk, idx) => 
                `<div class="word-chip" onclick="moveChunkToTarget(${idx})">${chunk}</div>`
            ).join("");

            targetBox.innerHTML = currentScrambleTarget.map((chunk, idx) => 
                `<div class="word-chip" onclick="moveChunkToSource(${idx})">${chunk}</div>`
            ).join("");

            document.getElementById("scramble-feedback").className = "feedback-box";
        }

        function moveChunkToTarget(idx) {
            const chunk = currentScrambleSource.splice(idx, 1)[0];
            currentScrambleTarget.push(chunk);
            renderScrambleUI();
        }

        function moveChunkToSource(idx) {
            const chunk = currentScrambleTarget.splice(idx, 1)[0];
            currentScrambleSource.push(chunk);
            renderScrambleUI();
        }

        function undoLastWord() {
            if (currentScrambleTarget.length > 0) {
                const chunk = currentScrambleTarget.pop();
                currentScrambleSource.push(chunk);
                renderScrambleUI();
            }
        }

        function resetScrambleSentence() {
            initScrambleSentence();
        }

        function checkScrambleSentence() {
            const data = scrambleData[gameState.scrambleIndex];
            const userSentence = currentScrambleTarget.join(" ");
            const feedback = document.getElementById("scramble-feedback");

            if (userSentence === data.full) {
                feedback.className = "feedback-box show correct";
                feedback.innerText = "Chính xác! Ghép câu rất tốt.";
                addScore(5);
                speakText(userSentence);
            } else {
                feedback.className = "feedback-box show incorrect";
                feedback.innerText = "Chưa chính xác, hãy thử kiểm tra lại thứ tự các cụm!";
            }
        }

        function listenScrambleSentence() {
            const data = scrambleData[gameState.scrambleIndex];
            speakText(data.full);
        }

        function navigateScramble(dir) {
            gameState.scrambleIndex += dir;
            if (gameState.scrambleIndex < 0) gameState.scrambleIndex = 0;
            if (gameState.scrambleIndex >= scrambleData.length) gameState.scrambleIndex = scrambleData.length - 1;
            initScrambleSentence();
        }

        function initGuidedWriting() {
            const container = document.getElementById("guided-writing-container");
            container.innerHTML = guidedWritingData.map((item, idx) => `
                <div style="background:#fff; padding:15px; border-radius:8px; margin-bottom:15px; border:1px solid #e2e8f0;">
                    <p style="font-weight:bold; margin-bottom:8px;">${idx + 1}. ${item.prompt}</p>
                    <textarea id="guided-input-${item.id}" class="writing-input" placeholder="Nhập câu tiếng Anh của bạn tại đây...">${gameState.guidedAnswers[item.id] || ''}</textarea>
                    <div style="display:flex; gap:10px;">
                        <button class="btn btn-success btn-sm" onclick="checkGuidedWriting('${item.id}', '${item.required}')">✔️ Chấm điểm</button>
                        <button class="btn btn-secondary btn-sm" onclick="speakText(document.getElementById('guided-input-${item.id}').value)">🔊 Nghe câu đã viết</button>
                    </div>
                    <div id="guided-feedback-${item.id}" class="feedback-box"></div>
                </div>
            `).join("");
        }

        function checkGuidedWriting(id, requiredPhrase) {
            const inputVal = document.getElementById(`guided-input-${id}`).value.trim();
            const feedback = document.getElementById(`guided-feedback-${id}`);
            gameState.guidedAnswers[id] = inputVal;

            let currentPoints = 0;
            let notes = [];

            if (inputVal.length === 0) {
                feedback.className = "feedback-box show incorrect";
                feedback.innerText = "Vui lòng nhập câu trả lời!";
                return;
            }

            if (inputVal.toLowerCase().includes(requiredPhrase.toLowerCase())) {
                currentPoints += 2;
            } else {
                notes.push(`Thiếu cụm từ bắt buộc: "${requiredPhrase}"`);
            }

            if (/^[A-Z]/.test(inputVal)) {
                currentPoints += 1;
            } else {
                notes.push("Chữ cái đầu câu phải viết hoa.");
            }

            if (/[.!?]$/.test(inputVal)) {
                currentPoints += 1;
            } else {
                notes.push("Cuối câu phải có dấu chấm hoặc dấu câu hợp lệ.");
            }

            if (inputVal.split(/\s+/).length >= 5) {
                currentPoints += 1;
            } else {
                notes.push("Câu cần có ít nhất 5 từ.");
            }

            if (!gameState.guidedScores[id]) {
                gameState.guidedScores[id] = currentPoints;
                addScore(currentPoints);
            }

            feedback.className = `feedback-box show ${currentPoints >= 4 ? 'correct' : 'incorrect'}`;
            feedback.innerHTML = `Đạt <strong>${currentPoints}/5 điểm</strong>.<br>` + (notes.length > 0 ? notes.join("<br>") : "Câu viết xuất sắc!");
            
            gameState.completedTasks.writing = true;
            saveGameState();
        }

        /* QUIZ TASK */
        function renderQuizQuestion() {
            const q = quizData[gameState.quizIndex];
            document.getElementById("quiz-counter").innerText = `${gameState.quizIndex + 1}/${quizData.length}`;
            const container = document.getElementById("quiz-container");
            const isAnswered = gameState.answeredQuiz[q.id] !== undefined;

            let optionsHTML = q.options.map((opt, idx) => {
                let btnClass = "option-btn";
                if (isAnswered) {
                    if (idx === q.correct) btnClass += " correct";
                    else if (gameState.answeredQuiz[q.id] === idx) btnClass += " incorrect";
                }
                return `<button class="${btnClass}" onclick="handleQuizAnswer('${q.id}', ${idx})" ${isAnswered ? 'disabled' : ''}>${String.fromCharCode(65 + idx)}. ${opt}</button>`;
            }).join("");

            container.innerHTML = `
                <div style="font-size:1.1rem; font-weight:bold; margin-bottom:10px;">${q.question}</div>
                <button class="btn btn-secondary btn-sm" style="margin-bottom:15px;" onclick="speakText('${q.question.replace(/'/g, "\\'")}')">🔊 Nghe câu hỏi</button>
                <div class="options-grid">${optionsHTML}</div>
                <div id="quiz-feedback" class="feedback-box ${isAnswered ? 'show ' + (gameState.answeredQuiz[q.id] === q.correct ? 'correct' : 'incorrect') : ''}">
                    ${isAnswered ? (gameState.answeredQuiz[q.id] === q.correct ? 'Chính xác!' : 'Chưa đúng. ' + q.explanation + '<br><em>Trích dẫn: "' + q.quote + '"</em>') : ''}
                </div>
            `;
        }

        function handleQuizAnswer(id, chosenIdx) {
            if (gameState.answeredQuiz[id] !== undefined) return;

            const q = quizData.find(item => item.id === id);
            gameState.answeredQuiz[id] = chosenIdx;

            if (chosenIdx === q.correct) {
                addScore(4);
            } else {
                if (!gameState.wrongQuizIds.includes(id)) {
                    gameState.wrongQuizIds.push(id);
                }
            }

            if (Object.keys(gameState.answeredQuiz).length === quizData.length) {
                gameState.completedTasks.quiz = true;
            }

            saveGameState();
            renderQuizQuestion();
        }

        function navigateQuiz(dir) {
            gameState.quizIndex += dir;
            if (gameState.quizIndex < 0) gameState.quizIndex = 0;
            if (gameState.quizIndex >= quizData.length) gameState.quizIndex = quizData.length - 1;
            renderQuizQuestion();
        }

        /* RESULTS SECTION */
        function renderResultsScreen() {
            const maxScore = 100;
            const percentage = Math.min(100, Math.round((gameState.score / maxScore) * 100));

            document.getElementById("final-score-text").innerText = `${gameState.score} Điểm (${percentage}%)`;

            const evalText = document.getElementById("final-eval-text");
            if (percentage >= 80) {
                evalText.style.color = "var(--secondary-color)";
                evalText.innerText = "🌟 Đã nắm bài tốt. Hãy đọc lại toàn bài và tự viết một đoạn tóm tắt 5–7 câu.";
            } else if (percentage >= 60) {
                evalText.style.color = "var(--warning-color)";
                evalText.innerText = "👍 Đã hiểu phần lớn nội dung. Cần luyện lại từ và phần viết câu.";
            } else {
                evalText.style.color = "var(--error-color)";
                evalText.innerText = "📚 Cần học lại theo thứ tự: bài đọc ➔ từ vựng ➔ cụm từ ➔ viết câu ➔ đọc hiểu.";
            }

            const reviewContainer = document.getElementById("wrong-items-review");
            let wrongHTML = "<h3>Danh sách cần ôn lại:</h3>";

            if (gameState.wrongVocabIds.length > 0) {
                wrongHTML += "<p><strong>Từ vựng sai:</strong> " + gameState.wrongVocabIds.map(id => vocabData.find(v => v.id === id).word).join(", ") + "</p>";
            }
            if (gameState.wrongQuizIds.length > 0) {
                wrongHTML += "<p><strong>Câu hỏi đọc hiểu sai:</strong> " + gameState.wrongQuizIds.map(id => quizData.find(q => q.id === id).question).join("<br>") + "</p>";
            }

            if (gameState.wrongVocabIds.length === 0 && gameState.wrongQuizIds.length === 0) {
                wrongHTML += "<p style='color:var(--secondary-color);'>Không có từ hoặc câu hỏi nào làm sai!</p>";
            }

            reviewContainer.innerHTML = wrongHTML;
        }

        /* STATE MANAGEMENT */
        function addScore(pts) {
            gameState.score += pts;
            updateDashboardUI();
            saveGameState();
        }

        function updateDashboardUI() {
            document.getElementById("stat-score").innerText = gameState.score;
            
            const completedCount = Object.values(gameState.completedTasks).filter(Boolean).length;
            document.getElementById("stat-completed").innerText = `${completedCount}/5`;

            const progressPct = Math.round((completedCount / 5) * 100);
            document.getElementById("stat-progress").innerText = `${progressPct}%`;
            document.getElementById("main-progress-bar").style.width = `${progressPct}%`;
        }

        function saveGameState() {
            localStorage.setItem(STORAGE_KEY, JSON.stringify(gameState));
        }

        function loadGameState() {
            const saved = localStorage.getItem(STORAGE_KEY);
            if (saved) {
                try {
                    gameState = JSON.parse(saved);
                    document.getElementById("speed-slider").value = gameState.speechRate || 0.8;
                    document.getElementById("speed-label").innerText = (gameState.speechRate || 0.8).toFixed(2) + "×";
                    document.getElementById("phrase-pause-chk").checked = !!gameState.phrasePause;
                    updateDashboardUI();
                } catch (e) {
                    console.error("Error loading state", e);
                }
            }
        }

        function checkResumeModal() {
            const saved = localStorage.getItem(STORAGE_KEY);
            if (saved) {
                document.getElementById("resume-modal").classList.add("active");
            } else {
                renderHomeScreen();
            }
        }

        function resumePreviousSession() {
            document.getElementById("resume-modal").classList.remove("active");
            loadGameState();
            renderHomeScreen();
        }

        function startFreshSession() {
            document.getElementById("resume-modal").classList.remove("active");
            localStorage.removeItem(STORAGE_KEY);
            renderHomeScreen();
        }

        function confirmResetProgress() {
            if (confirm("Bạn có chắc chắn muốn xóa toàn bộ tiến độ và làm lại từ đầu không?")) {
                localStorage.removeItem(STORAGE_KEY);
                location.reload();
            }
        }
